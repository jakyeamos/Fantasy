    **Step 2 — Create backend/src/fantasy/trends/similarity.py:**
    Algorithm: position-filtered Euclidean distance on 5 normalized features.
    ```python
    from __future__ import annotations
    import duckdb
    from fantasy.trends.models import SimilarPlayer

    SIMILARITY_COMPONENTS = [
        "comp_age_curve", "comp_ceiling", "comp_floor",
        "comp_role_stability", "comp_short_term",
    ]
    ADP_TIER_BINS = [(1, 6), (7, 12), (13, 24), (25, 48), (49, 200)]

    def _adp_tier(adp: float | None) -> int:
        if adp is None:
            return 5
        for i, (lo, hi) in enumerate(ADP_TIER_BINS):
            if lo <= adp <= hi:
                return i
        return 5

    def find_similar_players(
        player_id: str, conn: duckdb.DuckDBPyConnection, n: int = 3
    ) -> list[SimilarPlayer]:
        target = conn.execute(
            f"""SELECT p.position, t.startup_adp, {', '.join('t.' + c for c in SIMILARITY_COMPONENTS)}
                FROM player_trends t JOIN players p ON t.player_id = p.player_id
                WHERE t.player_id = ? ORDER BY t.season DESC LIMIT 1""",
            [player_id]
        ).fetchone()
        if target is None:
            return []
        position, t_adp = target[0], target[1]
        t_tier = _adp_tier(t_adp)
        t_feats = list(target[2:])

        candidates = conn.execute(
            f"""SELECT t.player_id, p.full_name, p.age, t.startup_adp,
                       {', '.join('t.' + c for c in SIMILARITY_COMPONENTS)}
                FROM player_trends t JOIN players p ON t.player_id = p.player_id
                WHERE t.player_id != ? AND p.position = ?
                ORDER BY t.season DESC""",
            [player_id, position]
        ).fetchall()

        scored = []
        for row in candidates:
            cand_id, cand_name, cand_age = row[0], row[1], row[2]
            c_tier = _adp_tier(row[3])
            if abs(c_tier - t_tier) > 1:
                continue
            c_feats = list(row[4:])
            dist = sum((a - b) ** 2 for a, b in zip(t_feats, c_feats)) ** 0.5
            scored.append((dist, cand_id, cand_name, cand_age, row[3]))

        if not scored:
            return []
        max_dist = max(s[0] for s in scored) or 1.0
        scored.sort(key=lambda x: x[0])

        results = []
        for dist, cid, cname, cage, cadp in scored[:n]:
            sim = max(0.0, 1.0 - dist / max_dist)
            context = f"age {cage} {position}, similar component profile and ADP tier"
            results.append(SimilarPlayer(
                player_id=cid, player_name=cname, similarity_score=round(sim, 2),
                archetype_label=None, context=context
            ))
        return results
    ```

    ```python
    # Confidence multipliers (D-05: impact_score = gap_magnitude × confidence_multiplier)
    CONFIDENCE_MULTIPLIERS = {"HIGH": 1.0, "MEDIUM": 0.7, "LOW": 0.4}

    CONFLICT_MATRIX: dict[tuple[str, str], bool] = {

        ("ignore_false_discount", "will_fall"): False,

    }

    CALENDAR_ESCALATION_STATES: dict[str, str] = {

    }
    ```

    `suggested_action` logic (D-13, veteran buy-low override):
    - If trend == "will_fall" AND adp_gap > 0 → "sell"
    - If trend == "will_rise" AND adp_gap < 0 → "buy"
    - If trend == "will_fall" AND adp_gap < 0 AND direction_label contains "contend" → "buy" (veteran buy-low)
    - Otherwise → "hold"

    `why_summary` template:
    - "will_rise": "Component trajectory is positive — {top_improving_component} trending up."
    - "will_fall": "Component trajectory is declining — {top_declining_component} falling."
    - "will_maintain": "Component scores are stable season-over-season."
