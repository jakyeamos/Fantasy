    **SimilarPlayersSection.tsx:**
    Collapsed by default. ChevronDown/ChevronUp toggle. 160ms ease transition.
    ```tsx
    import { useState } from "react"
    import { ChevronDown, ChevronUp } from "lucide-react"
    import type { SimilarPlayer } from "@/api/types"

    export function SimilarPlayersSection({ players }: { players: SimilarPlayer[] }) {
      const [open, setOpen] = useState(false)
      if (players.length === 0) return null
      return (
        <div className="mt-3">
          <button
            onClick={() => setOpen((o) => !o)}
            className="flex items-center gap-1.5 text-left"
            type="button"
          >
            <span className="terminal-label text-muted-foreground">
              SIMILAR PLAYERS ({players.length})
            </span>
            {open
              ? <ChevronUp className="size-3 text-muted-foreground" />
              : <ChevronDown className="size-3 text-muted-foreground" />}
          </button>
          {open && (
            <div className="mt-2 space-y-2" style={{ transition: "all 160ms ease" }}>
              {players.map((p) => (
                <div key={p.player_id} className="flex items-start justify-between gap-3">
                  <div>
                    <span className="text-sm font-bold text-foreground">{p.player_name}</span>
                    <p className="text-xs text-muted-foreground">{p.context}</p>
                  </div>
                  <span className="font-mono text-xs text-muted-foreground">
                    {(p.similarity_score * 100).toFixed(0)}%
                  </span>
                </div>
              ))}
            </div>
          )}
        </div>
      )
    }
    ```

    ```tsx
    import type { OpportunityFeedItem } from "@/api/types"
    import { CalendarEscalationLabel } from "./CalendarEscalationLabel"
    import { OwnershipSymbol } from "./OwnershipSymbol"
    import { SuggestedActionBadge } from "./SuggestedActionBadge"
    import { TrendBadge } from "./TrendBadge"
    import { ConfidenceIndicator } from "./ConfidenceIndicator"
    import { ConflictExplanationPanel } from "./ConflictExplanationPanel"
    import { SimilarPlayersSection } from "./SimilarPlayersSection"

    export function OpportunityCard({ item }: { item: OpportunityFeedItem }) {
      const adpGapDisplay = item.adp_gap >= 0
        ? `ADP Gap: +${item.adp_gap.toFixed(0)}`
        : `ADP Gap: ${item.adp_gap.toFixed(0)}`
      const impactDisplay = `IMPACT ${item.impact_score.toFixed(0)}`

      return (
        <div className="rounded-xl border border-border/60 bg-card/75 p-5">
          {/* Top row */}
          <div className="flex items-start justify-between gap-3">
            <div className="space-y-1">
              <CalendarEscalationLabel label={item.calendar_escalation_label} />
              <div className="flex items-center gap-2">
                <span className="text-sm font-bold text-foreground">{item.player_name}</span>
                <Badge variant="outline" className="text-xs">{item.position}</Badge>
                <OwnershipSymbol leagueIds={item.owned_in_leagues} />
              </div>
            </div>
            <SuggestedActionBadge action={item.suggested_action} />
          </div>

          {/* Second row */}
          <div className="mt-2 flex flex-wrap items-center gap-3">
            <TrendBadge label={item.trend_label} />
            <span className="font-mono text-xs text-muted-foreground">{adpGapDisplay}</span>
            <span className="terminal-label text-muted-foreground">{impactDisplay}</span>

          </div>

          {/* Body */}
          <p className="mt-3 text-sm leading-relaxed text-muted-foreground">{item.why_summary}</p>

          {/* LOW confidence annotation */}
          {item.trend_confidence === "LOW" && (

              Limited projection confidence. Treat as directional.
            </p>
          )}

          <ConflictExplanationPanel explanation={item.conflict_explanation} />

          {/* Similar players (collapsed by default) */}
          <SimilarPlayersSection players={item.similar_players} />
        </div>
      )
    }
    ```

    **OpportunityCardList.tsx:**
    Handles loading (5 skeleton cards), error, empty state, and full list. `space-y-4` between cards.
    ```tsx
    import { Skeleton } from "@/components/ui/skeleton"
    import type { OpportunityFeedItem } from "@/api/types"
    import { OpportunityCard } from "./OpportunityCard"

    function OpportunityCardSkeleton() {
      return (
        <div className="rounded-xl border border-border/60 bg-card/75 p-5 space-y-3">
          <Skeleton className="h-6 w-32" />
          <Skeleton className="h-4 w-48" />
          <Skeleton className="h-12 w-full" />
        </div>
      )
    }

    interface Props {
      items: OpportunityFeedItem[]
      isLoading: boolean
      isError: boolean
    }

    export function OpportunityCardList({ items, isLoading, isError }: Props) {
      if (isLoading) {
        return (
          <div className="space-y-4">
            {Array.from({ length: 5 }).map((_, i) => <OpportunityCardSkeleton key={i} />)}
          </div>
        )
      }
      if (isError) {
        return (
          <p className="text-sm text-muted-foreground">
            Opportunity feed could not be loaded. Check your data connection and try refreshing.
          </p>
        )
      }
      if (items.length === 0) {
        return (
          <div className="space-y-1">
            <p className="text-sm font-bold text-foreground">No Opportunities Available</p>
            <p className="text-sm text-muted-foreground">
              Ingest fresh roster and valuation data to surface buy, sell, and hold signals across your leagues.
            </p>
          </div>
        )
      }
      return (
        <div className="space-y-4">
          {items.map((item) => <OpportunityCard key={item.player_id} item={item} />)}
        </div>
      )
    }
    ```
