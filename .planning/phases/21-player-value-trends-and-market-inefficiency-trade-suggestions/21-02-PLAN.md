    **Step 4 — Create backend/src/fantasy/trends/trend_repo.py:**
    ```python
    from __future__ import annotations
    import duckdb

    COMPONENT_COLS = [
        "comp_current_production", "comp_short_term", "comp_role_stability",
        "comp_age_curve", "comp_insulation", "comp_market_liquidity",
        "comp_positional_scarcity", "comp_fragility", "comp_ceiling",
        "comp_floor", "comp_rerollability", "comp_contract",
    ]

    class TrendRepo:
        def __init__(self, conn: duckdb.DuckDBPyConnection) -> None:
            self._conn = conn

        def get_player_seasons(self, player_id: str) -> list[dict]:
            """Return rows ordered by season DESC, up to last 2 seasons."""
            rows = self._conn.execute(
                f"""SELECT season, {', '.join(COMPONENT_COLS)}, startup_adp, backfilled
                    FROM player_trends WHERE player_id = ?
                    ORDER BY season DESC LIMIT 2""",
                [player_id]
            ).fetchall()
            cols = ["season"] + COMPONENT_COLS + ["startup_adp", "backfilled"]
            return [dict(zip(cols, row)) for row in rows]

        def write_trend_row(
            self, player_id: str, season: int, trend_label: str,
            confidence: str, delta_magnitude: float, adp_delta: float | None,
            components: dict[str, float], startup_adp: float | None,
            backfilled: bool
        ) -> None:
            cols = ["player_id", "season", "trend_label", "confidence",
                    "delta_magnitude", "adp_delta", "startup_adp", "backfilled"] + COMPONENT_COLS
            vals = [player_id, season, trend_label, confidence,
                    delta_magnitude, adp_delta, startup_adp, backfilled] + \
                   [components.get(c) for c in COMPONENT_COLS]
            placeholders = ", ".join(["?"] * len(cols))
            self._conn.execute(
                f"""INSERT OR REPLACE INTO player_trends ({', '.join(cols)})
                    VALUES ({placeholders})""",
                vals
            )
    ```

    **Step 5 — Create backend/src/fantasy/trends/trend_engine.py:**
    ```python
    from __future__ import annotations
    import duckdb
    from fantasy.trends.models import TrendResult, TrendLabel, TrendConfidence
    from fantasy.trends.constants import COMPONENT_WEIGHTS, TREND_THRESHOLD, CONFIDENCE_THRESHOLDS
    from fantasy.trends.trend_repo import TrendRepo, COMPONENT_COLS

    class TrendEngine:
        def __init__(self, conn: duckdb.DuckDBPyConnection) -> None:
            self._conn = conn
            self._repo = TrendRepo(conn)

        def compute_trend(self, player_id: str) -> TrendResult:
            seasons = self._repo.get_player_seasons(player_id)
            if len(seasons) >= 2:
                current, prior = seasons[0], seasons[1]
                return self._compute_from_two_seasons(player_id, current, prior)
            elif len(seasons) == 1:
                return self._backfill_single_season(player_id, seasons[0])
            else:
                return self._backfill_no_history(player_id)

        def _compute_from_two_seasons(self, player_id, current, prior) -> TrendResult:
            component_deltas = {}
            weighted_delta = 0.0
            weight_sum = 0.0
            for comp, weight in COMPONENT_WEIGHTS.items():
                cur_val = current.get(comp) or 0.0
                pri_val = prior.get(comp) or 0.0
                delta = (cur_val - pri_val) * (1 if weight >= 0 else -1)
                component_deltas[comp] = cur_val - pri_val
                weighted_delta += delta * abs(weight)
                weight_sum += abs(weight)
            agg = weighted_delta / max(weight_sum, 1e-9)

            trend_label: TrendLabel = (
                "will_rise" if agg > TREND_THRESHOLD
                else "will_fall" if agg < -TREND_THRESHOLD
                else "will_maintain"
            )

            # Confidence: HIGH if both non-backfilled and components agree; LOW if backfilled
            backfilled = bool(prior.get("backfilled"))
            if backfilled:
                confidence: TrendConfidence = "LOW"
            else:
                # Measure agreement: fraction of weighted components moving in same direction as agg
                agreeing = sum(
                    1 for c, w in COMPONENT_WEIGHTS.items()
                    if (component_deltas.get(c, 0.0) * (1 if w >= 0 else -1)) * agg >= 0
                )
                agreement_ratio = agreeing / max(len(COMPONENT_WEIGHTS), 1)
                confidence = "HIGH" if agreement_ratio >= 0.7 else "MEDIUM"

            adp_prior = prior.get("startup_adp")
            adp_current = current.get("startup_adp")
            adp_delta = (adp_prior - adp_current) if (adp_prior is not None and adp_current is not None) else None

            return TrendResult(
                player_id=player_id,
                trend_label=trend_label,
                confidence=confidence,
                delta_magnitude=agg,
                component_deltas=component_deltas,
                adp_delta=adp_delta,
                seasons_compared=2,
                backfilled=backfilled,
            )

        def _backfill_single_season(self, player_id, current) -> TrendResult:
            # Single season: use component absolute values to infer trajectory
            age_curve = current.get("comp_age_curve") or 0.5
            short_term = current.get("comp_short_term") or 0.5
            heuristic = (age_curve * 0.5 + short_term * 0.5) - 0.5
            trend_label: TrendLabel = (
                "will_rise" if heuristic > TREND_THRESHOLD
                else "will_fall" if heuristic < -TREND_THRESHOLD
                else "will_maintain"
            )
            return TrendResult(
                player_id=player_id,
                trend_label=trend_label,
                confidence="LOW",
                delta_magnitude=heuristic,
                component_deltas={},
                adp_delta=None,
                seasons_compared=1,
                backfilled=True,
            )

        def _backfill_no_history(self, player_id) -> TrendResult:
            return TrendResult(
                player_id=player_id,
                trend_label="will_maintain",
                confidence="LOW",
                delta_magnitude=0.0,
                component_deltas={},
                adp_delta=None,
                seasons_compared=0,
                backfilled=True,
            )
    ```
