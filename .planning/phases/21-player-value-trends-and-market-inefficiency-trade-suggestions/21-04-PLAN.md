```python
    def _direction_lens(self, value: PlayerValue, direction_label: str) -> float:
        weights = DIRECTION_VALUE_WEIGHTS[direction_label]
        component_scores = {
            "current_production": value.comp_current_production or 0.0,
            # ... 12 components
        }
        total = 0.0
        weight_sum = 0.0
        for component, weight in weights.items():
            score = component_scores[component]
            total += (score if weight >= 0 else (1.0 - score)) * abs(weight)
            weight_sum += abs(weight)
        return _clamp01(total / max(weight_sum, 1e-9))
```

    ```python
    def _direction_lens(
        self,
        value: PlayerValue,
        direction_label: str,
        trend_result: TrendResult | None = None,
    ) -> float:
        # ... existing weighted dot-product (unchanged) ...
        raw_score = _clamp01(total / max(weight_sum, 1e-9))

        # D-11: weaken stabilization prior for elite will_fall players
        if trend_result is not None and trend_result.trend_label == "will_fall":
            if self._is_elite(value):
                reduction = {"HIGH": 0.70, "MEDIUM": 0.40, "LOW": 0.15}.get(
                    trend_result.confidence, 0.0
                )
                raw_score = raw_score * (1.0 - reduction)

        return _clamp01(raw_score)

    def _is_elite(self, value: PlayerValue) -> bool:
        """Elite threshold: composite of insulation + ceiling + floor above 0.70."""
        insulation = value.comp_insulation or 0.0
        ceiling = value.comp_ceiling or 0.0
        floor = value.comp_floor or 0.0
        return (insulation + ceiling + floor) / 3.0 >= 0.70
    ```

    **Step 4 — Add trend_to_supporting_factor() to backend/src/fantasy/trends/models.py** (read the file first, then append the function):
    ```python
    def _confidence_multiplier(confidence: TrendConfidence) -> float:
        return {"HIGH": 1.0, "MEDIUM": 0.7, "LOW": 0.4}.get(confidence, 0.4)

    def trend_to_supporting_factor(trend: TrendResult) -> dict:
        direction_map: dict[str, str] = {
            "will_rise": "positive",
            "will_fall": "negative",
            "will_maintain": "neutral",
        }
        magnitude = round(
            abs(trend.delta_magnitude) * _confidence_multiplier(trend.confidence), 2
        )
        direction_words = {
            "will_rise": "rise",
            "will_fall": "fall",
            "will_maintain": "remain stable",
        }
        explanation = (
            f"Component scores project a {direction_words[trend.trend_label]} next season "
            f"({trend.seasons_compared} season{'s' if trend.seasons_compared != 1 else ''} compared"
            f"{', backfilled' if trend.backfilled else ''})."
        )
        return {
            "factor_name": "Value Trend",
            "direction": direction_map[trend.trend_label],
            "magnitude": magnitude,
            "explanation": explanation,
        }
    ```
