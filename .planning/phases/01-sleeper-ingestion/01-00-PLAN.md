```python
# leagues: INGEST-01
leagues(
    league_id VARCHAR PRIMARY KEY,
    name VARCHAR NOT NULL,
    season VARCHAR NOT NULL,
    scoring_settings VARCHAR NOT NULL,   # JSON blob stored as VARCHAR
    roster_positions VARCHAR NOT NULL,   # JSON blob stored as VARCHAR
    settings_blob VARCHAR,               # raw settings object
    superflex BOOLEAN NOT NULL DEFAULT FALSE,
    tep BOOLEAN NOT NULL DEFAULT FALSE,
    ppr FLOAT NOT NULL DEFAULT 0.0,
    ingested_at TIMESTAMP NOT NULL DEFAULT current_timestamp
)

# rosters: INGEST-02
rosters(
    id INTEGER PRIMARY KEY,             # DuckDB auto-increment
    league_id VARCHAR NOT NULL,
    roster_id INTEGER NOT NULL,
    owner_id VARCHAR,
    starters VARCHAR NOT NULL,          # JSON list
    players VARCHAR NOT NULL,           # JSON list (all players incl bench)
    reserve VARCHAR,                    # JSON list (IR)
    taxi VARCHAR,                       # JSON list (taxi squad)
    ingested_at TIMESTAMP NOT NULL DEFAULT current_timestamp,
    UNIQUE(league_id, roster_id)
)

# standings: INGEST-03
standings(
    id INTEGER PRIMARY KEY,
    league_id VARCHAR NOT NULL,
    roster_id INTEGER NOT NULL,
    wins INTEGER NOT NULL DEFAULT 0,
    losses INTEGER NOT NULL DEFAULT 0,
    ties INTEGER NOT NULL DEFAULT 0,
    fpts FLOAT NOT NULL DEFAULT 0.0,
    fpts_against FLOAT NOT NULL DEFAULT 0.0,
    ingested_at TIMESTAMP NOT NULL DEFAULT current_timestamp,
    UNIQUE(league_id, roster_id)
)

# traded_picks: INGEST-04
traded_picks(
    id INTEGER PRIMARY KEY,
    league_id VARCHAR NOT NULL,
    season VARCHAR NOT NULL,
    round INTEGER NOT NULL,
    roster_id INTEGER NOT NULL,          # original roster (previous owner's roster)
    owner_id VARCHAR NOT NULL,           # current holder roster_id
    previous_owner_id VARCHAR,
    ingested_at TIMESTAMP NOT NULL DEFAULT current_timestamp,
    UNIQUE(league_id, season, round, roster_id)
)

# transactions: INGEST-05, INGEST-06
transactions(
    transaction_id VARCHAR PRIMARY KEY,
    league_id VARCHAR NOT NULL,
    type VARCHAR NOT NULL,               # 'trade', 'free_agent', 'waiver'
    status VARCHAR NOT NULL,
    created_at TIMESTAMP,
    roster_ids VARCHAR,                  # JSON list
    adds VARCHAR,                        # JSON dict player_id -> roster_id
    drops VARCHAR,                       # JSON dict player_id -> roster_id
    draft_picks VARCHAR,                 # JSON list of pick objects
    week INTEGER,
    ingested_at TIMESTAMP NOT NULL DEFAULT current_timestamp
)

# players: cache for /v1/players/nfl (5MB payload — refresh separately)
players(
    player_id VARCHAR PRIMARY KEY,
    full_name VARCHAR,
    position VARCHAR,
    team VARCHAR,
    age INTEGER,
    metadata_blob VARCHAR,              # full raw JSON
    refreshed_at TIMESTAMP NOT NULL DEFAULT current_timestamp
)

# ingest_runs: INGEST-08 cursor tracking
ingest_runs(
    id INTEGER PRIMARY KEY,
    league_id VARCHAR NOT NULL,
    run_type VARCHAR NOT NULL,           # 'full', 'incremental'
    status VARCHAR NOT NULL,             # 'running', 'complete', 'failed'
    started_at TIMESTAMP NOT NULL DEFAULT current_timestamp,
    completed_at TIMESTAMP,
    cursor_json VARCHAR,                 # JSON: type-specific state cursors
    gaps_json VARCHAR,                   # JSON: array of gap objects
    error_message VARCHAR
)

# corrections: INGEST-07
corrections(
    id INTEGER PRIMARY KEY,
    league_id VARCHAR NOT NULL,
    entity_type VARCHAR NOT NULL,        # 'player', 'pick', 'roster_slot', 'league_setting'
    entity_id VARCHAR NOT NULL,
    field VARCHAR NOT NULL,
    original_value VARCHAR,
    corrected_value VARCHAR NOT NULL,
    corrected_by VARCHAR NOT NULL DEFAULT 'user',
    created_at TIMESTAMP NOT NULL DEFAULT current_timestamp,
    UNIQUE(league_id, entity_type, entity_id, field)
)
```

Use DuckDB `INTEGER PRIMARY KEY` (auto-increments natively). Do NOT use PostgreSQL `SERIAL` type — DuckDB does not support it (see RESEARCH.md pitfalls).
