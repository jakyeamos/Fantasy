```typescript
import { queryOptions } from "@tanstack/react-query"
import type { PortfolioExposureResponse, RecalibrationHealth, SnapshotAnchor, DiffRow } from "./types"

// staleTime: 5 min
export const portfolioExposureOptions = () =>
  queryOptions({
    queryKey: ["portfolio", "exposure"],
    queryFn: async (): Promise<PortfolioExposureResponse> => {
      const res = await fetch("/api/portfolio/exposure")
      if (!res.ok) throw new Error("Failed to fetch portfolio exposure")
      return res.json()
    },
    staleTime: 5 * 60 * 1000,
  })

// staleTime: 1 hour
export const portfolioHealthOptions = () =>
  queryOptions({
    queryKey: ["portfolio", "health"],
    queryFn: async (): Promise<RecalibrationHealth> => {
      const res = await fetch("/api/portfolio/health")
      if (!res.ok) throw new Error("Failed to fetch portfolio health")
      return res.json()
    },
    staleTime: 60 * 60 * 1000,
  })

// staleTime: 1 min
export const snapshotAnchorsOptions = (leagueId: string) =>
  queryOptions({
    queryKey: ["snapshot-anchors", leagueId],
    queryFn: async (): Promise<SnapshotAnchor[]> => {
      const res = await fetch(`/api/leagues/${leagueId}/snapshot-anchors`)
      if (!res.ok) throw new Error("Failed to fetch snapshot anchors")
      return res.json()
    },
    staleTime: 60 * 1000,
  })

// staleTime: 5 min
export const snapshotDiffOptions = (leagueId: string, snapshotId: number, rosterId: number) =>
  queryOptions({
    queryKey: ["snapshot-diff", leagueId, snapshotId, rosterId],
    queryFn: async (): Promise<DiffRow[]> => {
      const res = await fetch(`/api/leagues/${leagueId}/snapshot-diff?snapshot_id=${snapshotId}&roster_id=${rosterId}`)
      if (!res.ok) throw new Error("Failed to fetch snapshot diff")
      return res.json()
    },
    staleTime: 5 * 60 * 1000,
    enabled: snapshotId > 0 && rosterId > 0,
  })
```
