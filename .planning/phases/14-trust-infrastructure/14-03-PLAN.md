Create `frontend/src/components/FormatWarningBanner.tsx`:

The component fetches scan + acknowledgment state, then renders:
- Nothing if scan has no partially_supported or unsupported entries
- A dismissible amber/warning banner for partially-supported entries (with "I understand, continue" button)
- A persistent non-dismissible red/destructive banner for unsupported entries

```tsx
import { useMutation, useQuery, useQueryClient } from "@tanstack/react-query"
import { AlertTriangle, XOctagon } from "lucide-react"

import {
  acknowledgeLeagueFormat,
  leagueAcknowledgedOptions,
  leagueFormatScanOptions,
} from "@/api/queries"
import type { RuleScanEntry } from "@/api/types"
import { Button } from "@/components/ui/button"

export function FormatWarningBanner({ leagueId }: { leagueId: string }) {
  const queryClient = useQueryClient()
  const scanQuery = useQuery(leagueFormatScanOptions(leagueId))
  const ackedQuery = useQuery(leagueAcknowledgedOptions(leagueId))

  const acknowledgeMutation = useMutation({
    mutationFn: () => acknowledgeLeagueFormat(leagueId),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ["trust", leagueId] })
    },
  })

  if (scanQuery.isLoading || scanQuery.isError || !scanQuery.data) return null

  const scan = scanQuery.data
  const isAcknowledged = ackedQuery.data?.acknowledged ?? false

  const unsupportedEntries = scan.entries.filter(
    (e) => e.support_level === "unsupported"
  )
  const partialEntries = scan.entries.filter(
    (e) => e.support_level === "partially_supported" && e.distorts_recommendations
  )

  // Nothing to show if no flags at all, or partial flags are already acknowledged
  const showPartial = partialEntries.length > 0 && !isAcknowledged
  const showUnsupported = unsupportedEntries.length > 0

  if (!showPartial && !showUnsupported) return null

  return (
    <div className="space-y-3">
      {showUnsupported && (
        <UnsupportedBanner entries={unsupportedEntries} />
      )}
      {showPartial && (
        <PartialSupportBanner
          entries={partialEntries}
          onAcknowledge={() => acknowledgeMutation.mutate()}
          isPending={acknowledgeMutation.isPending}
        />
      )}
    </div>
  )
}

function UnsupportedBanner({ entries }: { entries: RuleScanEntry[] }) {
  return (
    <div className="glass-panel rounded-xl border border-destructive/25 bg-destructive/10 p-5">
      <div className="flex items-start gap-3">
        <div className="flex size-10 items-center justify-center rounded-lg border border-destructive/20 bg-destructive/15 text-destructive">
          <XOctagon className="size-4" />
        </div>
        <div className="min-w-0 flex-1 space-y-2">
          <p className="terminal-label text-destructive">
            Unsupported format detected — advice may be incorrect
          </p>
          <ul className="space-y-1">
            {entries.map((e) => (
              <li key={e.rule} className="text-sm text-muted-foreground">
                <span className="font-medium capitalize">{e.rule.replace(/_/g, " ")}</span>
                {" — "}
                {e.reason}
              </li>
            ))}
          </ul>
          <p className="text-xs text-destructive/70">
            This warning cannot be dismissed. Proceed with caution.
          </p>
        </div>
      </div>
    </div>
  )
}

function PartialSupportBanner({
  entries,
  onAcknowledge,
  isPending,
}: {
  entries: RuleScanEntry[]
  onAcknowledge: () => void
  isPending: boolean
}) {
  return (
    <div className="glass-panel rounded-xl border border-primary/25 bg-card/65 p-5">
      <div className="flex items-start gap-3">
        <div className="flex size-10 items-center justify-center rounded-lg border border-primary/20 bg-primary/10 text-primary">
          <AlertTriangle className="size-4" />
        </div>
        <div className="min-w-0 flex-1 space-y-3">
          <div className="space-y-1">
            <p className="terminal-label text-primary/85">Partially supported format</p>
            <p className="text-sm text-muted-foreground">
              Some league rules are not fully modeled. Recommendations may be less accurate.
            </p>
          </div>
          <ul className="space-y-1">
            {entries.map((e) => (
              <li key={e.rule} className="text-sm text-muted-foreground">
                <span className="font-medium capitalize">{e.rule.replace(/_/g, " ")}</span>
                {" — "}
                {e.reason}
              </li>
            ))}
          </ul>
          <Button
            variant="outline"
            size="sm"
            onClick={onAcknowledge}
            disabled={isPending}
          >
            I understand, continue
          </Button>
        </div>
      </div>
    </div>
  )
}
```
