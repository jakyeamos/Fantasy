```python
from __future__ import annotations
import json
from datetime import datetime, timezone

import duckdb

from fantasy.trust.models import FormatAcknowledgment


class TrustRepo:
    def __init__(self, conn: duckdb.DuckDBPyConnection) -> None:
        self._conn = conn

    def upsert_acknowledgment(self, league_id: str, acknowledged_rules: list[str]) -> FormatAcknowledgment:
        now = datetime.now(timezone.utc).isoformat()
        rules_json = json.dumps(acknowledged_rules, separators=(",", ":"))
        # Check if row exists
        existing = self._conn.execute(
            "SELECT id FROM league_format_acknowledgments WHERE league_id = ?",
            [league_id],
        ).fetchone()
        if existing:
            self._conn.execute(
                """
                UPDATE league_format_acknowledgments
                SET acknowledged_rules = ?, updated_at = ?
                WHERE league_id = ?
                """,
                [rules_json, now, league_id],
            )
        else:
            next_id = int(
                self._conn.execute(
                    "SELECT COALESCE(MAX(id), 0) + 1 FROM league_format_acknowledgments"
                ).fetchone()[0]
            )
            self._conn.execute(
                """
                INSERT INTO league_format_acknowledgments
                    (id, league_id, acknowledged_rules, acknowledged_at, updated_at)
                VALUES (?, ?, ?, ?, ?)
                """,
                [next_id, league_id, rules_json, now, now],
            )
        return FormatAcknowledgment(
            league_id=league_id,
            acknowledged_rules=acknowledged_rules,
            acknowledged_at=now,
        )

    def get_acknowledgment(self, league_id: str) -> FormatAcknowledgment | None:
        row = self._conn.execute(
            """
            SELECT league_id, acknowledged_rules, acknowledged_at
            FROM league_format_acknowledgments
            WHERE league_id = ?
            """,
            [league_id],
        ).fetchone()
        if row is None:
            return None
        return FormatAcknowledgment(
            league_id=row[0],
            acknowledged_rules=json.loads(row[1] or "[]"),
            acknowledged_at=str(row[2]),
        )

    def clear_acknowledgment(self, league_id: str) -> None:
        self._conn.execute(
            "DELETE FROM league_format_acknowledgments WHERE league_id = ?",
            [league_id],
        )
```

```python
from __future__ import annotations
import json

import duckdb
from fastapi import APIRouter, Depends, HTTPException

from fantasy.ingestion.sleeper_mapper import LeagueSettings
from fantasy.routers.deps import get_read_db_conn, get_write_db_conn
from fantasy.trust.models import FormatAcknowledgment, LeagueFormatScan
from fantasy.trust.scanner import scan_league_format
from fantasy.trust.trust_repo import TrustRepo

router = APIRouter(prefix="/trust", tags=["trust"])


def _load_league_settings(league_id: str, conn: duckdb.DuckDBPyConnection) -> LeagueSettings:
    row = conn.execute(
        """
        SELECT league_id, name, season, scoring_settings, roster_positions,
               settings_blob, superflex, tep, ppr
        FROM leagues
        WHERE league_id = ?
        LIMIT 1
        """,
        [league_id],
    ).fetchone()
    if row is None:
        raise HTTPException(status_code=404, detail=f"League not found: {league_id}")
    return LeagueSettings(
        league_id=row[0],
        name=row[1],
        season=row[2],
        scoring_settings=json.loads(row[3] or "{}"),
        roster_positions=json.loads(row[4] or "[]"),
        settings_blob=json.loads(row[5] or "{}"),
        superflex=bool(row[6]),
        tep=bool(row[7]),
        ppr=float(row[8]),
    )


@router.get("/{league_id}/scan", response_model=LeagueFormatScan)
def get_league_scan(
    league_id: str,
    conn: duckdb.DuckDBPyConnection = Depends(get_read_db_conn),
) -> LeagueFormatScan:
    settings = _load_league_settings(league_id, conn)
    return scan_league_format(settings)


@router.post("/{league_id}/acknowledge")
def acknowledge_league(
    league_id: str,
    conn: duckdb.DuckDBPyConnection = Depends(get_write_db_conn),
) -> dict:
    settings = _load_league_settings(league_id, conn)
    scan = scan_league_format(settings)
    current_rules = frozenset(e.rule for e in scan.entries)
    repo = TrustRepo(conn)
    # Invalidate if existing ack covers a different rule set
    existing = repo.get_acknowledgment(league_id)
    if existing is not None:
        prior_rules = frozenset(existing.acknowledged_rules)
        if prior_rules != current_rules:
            repo.clear_acknowledgment(league_id)
    ack = repo.upsert_acknowledgment(league_id, list(current_rules))
    return {"league_id": league_id, "acknowledgment": ack.model_dump()}


@router.get("/{league_id}/acknowledged")
def get_acknowledged(
    league_id: str,
    conn: duckdb.DuckDBPyConnection = Depends(get_read_db_conn),
) -> dict:
    repo = TrustRepo(conn)
    ack = repo.get_acknowledgment(league_id)
    return {
        "league_id": league_id,
        "acknowledged": ack is not None,
        "acknowledgment": ack.model_dump() if ack else None,
    }
```

def test_scan_returns_median_wins_entry(client):

    data = resp.json()
    rules = [e["rule"] for e in data["entries"]]
    assert "median_wins" in rules
    entry = next(e for e in data["entries"] if e["rule"] == "median_wins")
    assert entry["support_level"] == "partially_supported"
    assert data["needs_acknowledgment"] is True


def test_acknowledge_and_check(client):

    assert resp.status_code == 200

    # Check acknowledged state

    assert resp2.status_code == 200
    data = resp2.json()
    assert data["acknowledged"] is True
    assert data["acknowledgment"] is not None
    assert len(data["acknowledgment"]["acknowledged_rules"]) > 0
