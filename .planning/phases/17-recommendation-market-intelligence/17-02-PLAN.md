```python
from __future__ import annotations

ELITE_INSULATION_THRESHOLD: float = 0.70
ELITE_CEILING_THRESHOLD: float = 0.65
ELITE_FLOOR_THRESHOLD: float = 0.50
ELITE_AGE_CURVE_THRESHOLD: float = 0.50
BEARISH_PRODUCTION_FLOOR: float = 0.40
BULLISH_CEILING_CAP: float = 0.65
LOW_INSULATION_THRESHOLD: float = 0.40
BREAKOUT_CEILING_THRESHOLD: float = 0.80


def is_elite(value: "PlayerValue") -> bool:  # type: ignore[name-defined]
    """True when player meets all 4 elite composite thresholds (D-08)."""
    return (
        (value.comp_insulation or 0.0) >= ELITE_INSULATION_THRESHOLD
        and (value.comp_ceiling or 0.0) >= ELITE_CEILING_THRESHOLD
        and (value.comp_floor or 0.0) >= ELITE_FLOOR_THRESHOLD
        and (value.comp_age_curve or 0.0) >= ELITE_AGE_CURVE_THRESHOLD
    )


def apply_stabilization(value: "PlayerValue", active_flag_types: list[str]) -> "PlayerValue":
    """Apply bidirectional anti-overreaction priors. Silent — no annotation (D-09).

    Bearish dampening: elite player with low production gets floor pull.
    Exception: bypassed if injury_recovery or role_compression flag is active (real signal overrides prior).
    Bullish dampening: low-insulation breakout candidate gets ceiling capped.
    """
    bypass_flags = {"injury_recovery", "role_compression"}
    if is_elite(value) and not bypass_flags.intersection(active_flag_types):
        if (value.comp_current_production or 0.0) < BEARISH_PRODUCTION_FLOOR:
            value.comp_current_production = BEARISH_PRODUCTION_FLOOR

    if (value.comp_insulation or 0.0) <= LOW_INSULATION_THRESHOLD:
        if (value.comp_ceiling or 0.0) >= BREAKOUT_CEILING_THRESHOLD:
            value.comp_ceiling = BULLISH_CEILING_CAP

    return value
```

Note: Use string annotations (`"PlayerValue"`) to avoid circular imports — anti_overreaction.py must not import from intelligence/valuation_engine.py; callers pass PlayerValue objects in. Add `from __future__ import annotations` at top so runtime does not need the actual type at definition time.

Logic per RESEARCH.md Pattern 4:
- injury_recovery: `player_dict.get("injury_status") in ("IR", "Out", "Doubtful")`
- depth_chart_competition: `player_dict.get("depth_chart_order", 1) >= 2` and position in ("QB","RB","WR","TE")

- role_compression: stored_player was 1 and now >=2
- team_change: `player_dict.get("team") != stored_player.get("team")` when stored_player is not None; expires_at = datetime.utcnow() + timedelta(days=30)
- age_cliff_proximity: POSITIONAL_CLIFF_AGES.get(position) and (cliff_age - player_age) <= 1.0 AND >= 0.0 (within 1 year before cliff); no expiry

For role change flags (expansion/compression), set expires_at=None (re-derived on every ingest).
For injury_recovery, set expires_at=None (expires when injury_status changes on next ingest).
