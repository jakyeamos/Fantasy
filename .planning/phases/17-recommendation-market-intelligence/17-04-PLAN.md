`build_trade_card(evaluation: "TradeEvaluation", league_id: str, user_roster_id: int) → list[RecommendationCard]`:
- Derive overall verdict from evaluation's dimension scores (market_fairness, direction_fit, roster_fit)
- Create primary card: recommendation_type="trade", priority_rank=compute_priority(direction_fit.score, max(dim.score for dim...), 0.7, evaluation.timing_quality.score), headline="Trade Advances Your Direction" or "Trade Undermines Direction" based on direction_fit.score vs 0.5, action="Evaluate the proposed trade.", why_summary from dimension scores, supporting_factors=[one factor per dimension], confidence_label=confidence_label_from_score(mean of dimension confidence floats), confidence_score=mean, downside_of_inaction="Missing the timing window on this deal reduces leverage.", what_would_change_this_call="A better market offer or change in direction label.", horizon="30_days", cta_label="Evaluate This Trade", cta_destination=f"/league/{league_id}/trade"
- Return [primary_card]

`build_hygiene_card(suggestion: "HygieneSuggestion", league_id: str) → RecommendationCard`:
- recommendation_type maps: "consolidate"→"trade", "cut"→"drop", "stash"→"stash", "taxi"→"taxi"; default "hold"
- target_entity_ids = suggestion.primary_player_ids
- headline = f"{suggestion.action_type.title()}: {', '.join(suggestion.primary_player_names[:2])}"
- why_summary = suggestion.reasoning
- priority_rank = compute_priority(suggestion.direction_fit_score, 0.7, 0.8, 0.6)
- cta_label="Review Roster", cta_destination=f"/league/{league_id}/overview"

`build_waiver_card(rec: "WaiverRecommendation", league_id: str) → RecommendationCard`:
- urgency_float = {"High": 0.9, "Medium": 0.6, "Low": 0.3}.get(rec.urgency, 0.5)
- recommendation_type = "bid" if rec.recommendation_label == "faab_bid" else "stash"
- headline = f"Add {rec.player_name} ({rec.position})"
- why_summary = rec.rationale
- priority_rank = compute_priority(0.7, 0.7, 0.8, urgency_float)
- cta_label="Check Waivers", cta_destination=f"/league/{league_id}/waivers"
