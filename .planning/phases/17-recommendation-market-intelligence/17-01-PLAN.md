```python
import duckdb
from fantasy.recommendation.models import RecommendationCard, SupportingFactor, ConfidenceLabel
from fantasy.recommendation.constants import confidence_label_from_score

def compute_priority(impact: float, confidence: float, execution: float, urgency: float) -> int:
    """Priority = impact × confidence × execution × urgency. Returns int 1–100 (1 = highest priority).
    D-05: uniform formula across all modules, same weights."""
    raw_score = impact * confidence * execution * urgency
    return max(1, min(100, round((1.0 - raw_score) * 99) + 1))

class RecommendationCardEngine:
    def __init__(self, conn: duckdb.DuckDBPyConnection) -> None:
        self._conn = conn
```
