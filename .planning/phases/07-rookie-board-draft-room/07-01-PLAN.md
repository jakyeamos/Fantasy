interface RookieBoardResponse {

  title: string
  description: string
}

    CREATE TABLE IF NOT EXISTS rookie_board_cache (

        class_strength_signal   FLOAT NOT NULL DEFAULT 0.0,
        board_json              VARCHAR NOT NULL,

    **`rookie_repo.py`:** Repository class following the `trade_repo.py` / `pick_repo.py` pattern. Constructor takes `duckdb.DuckDBPyConnection`.

    Methods:
    - `get_league_settings(league_id: str) -> dict`: Reads from `leagues` table — returns dict with `ppr`, `superflex`, `tep`, `roster_positions` fields.
    - `get_available_rookies(league_id: str) -> list[dict]`: Reads from `players` JOIN `player_adp_baseline` WHERE position IN ('WR', 'RB', 'QB', 'TE') AND adp IS NOT NULL, ORDER BY adp ASC. Returns list of dicts with player_id, full_name, position, age, team, adp.
    - `get_league_tendencies(league_id: str) -> list[dict]`: Reads from `league_draft_tendencies` WHERE league_id = ?. Returns list of dicts.
    - `save_board_cache(league_id: str, class_strength_signal: float, board_json: str) -> None`: Upserts into `rookie_board_cache` (INSERT OR REPLACE on league_id unique constraint).
    - `get_cached_class_strength(league_id: str) -> float`: Reads `class_strength_signal` from `rookie_board_cache` WHERE league_id = ?. Returns 0.0 if no cache entry.
    - `save_tendencies(league_id: str, tendencies: list[dict]) -> None`: Deletes existing tendencies for league_id, inserts new batch.
    - `get_pick_slots_for_league(league_id: str) -> list[dict]`: Reads from `traded_picks` WHERE league_id = ? to get available pick slots. Returns list with pick slot info.
