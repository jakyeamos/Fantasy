```python
import nflreadpy as nfl

class NflReadPyLoader:
    def load_combine(self, start_year: int = 2011, end_year: int = 2026) -> pl.DataFrame:
        """Load combine data. Returns Polars DataFrame with player_name, pos, ht, wt, forty, vertical, cone, shuttle, bench, draft_ovr, draft_round, draft_year."""
        combine_df = nfl.load_combine(seasons=list(range(start_year, end_year)))
        return pl.from_pandas(combine_df)

    def load_players(self) -> pl.DataFrame:
        """Load player bio + draft info. Returns gsis_id, display_name, birth_date, college_name, rookie_season, draft_year, draft_round, draft_pick."""
        players_df = nfl.load_players()
        return pl.from_pandas(players_df)

    def load_draft_picks(self, start_year: int = 2011, end_year: int = 2026) -> pl.DataFrame:
        """Load draft picks. Returns season, round, pick, gsis_id, pfr_id, position."""
        draft_df = nfl.load_draft_picks(seasons=list(range(start_year, end_year)))
        return pl.from_pandas(draft_df)
```

```python
class FeatureBuilder:
    def __init__(self, prospect_repo: ProspectRepo):
        self.repo = prospect_repo

    def build_id_map(self, players_df: pl.DataFrame) -> pl.DataFrame:
        """Build canonical ID map from nflreadpy load_players() output. Columns: gsis_id, pfr_id, display_name, birth_date, draft_year."""

    def compute_seasonal_finish_rank(self, weekly_stats_df: pl.DataFrame, position: str, season: int) -> pl.DataFrame:
        """Roll up weekly stats to season totals, rank within position, return player_id + rank + fantasy_points."""

    def label_outcome(self, position: str, season_ranks: list[int]) -> str | None:
        """Apply position-specific hit/mediocre/bust thresholds from constants.
        - season_ranks: list of positional finish ranks for each post-draft season
        - Returns None if len(season_ranks) < MIN_OUTCOME_SEASONS
        - Uses HIT_THRESHOLDS[position] and MEDIOCRE_THRESHOLDS[position]
        - Per D-03: timing NOT factored — counts qualifying seasons regardless of when they occur
        """

    def build_feature_vector(self, combine_row: dict, draft_row: dict, player_bio: dict) -> ProspectFeatures:
        """Construct a ProspectFeatures from pre-draft data ONLY.
        CRITICAL: No NFL career stats in the feature vector — this prevents look-ahead bias (Research pitfall #1).
        - age_at_draft = draft_date - birth_date
        - draft_ovr from draft_picks
        - Athletic testing from combine (forty, vertical, bench, cone, shuttle, weight, height)
        - College production proxies set to None if CFB data unavailable (graceful degradation per research)
        """

    def build_all_features(self, combine_df, draft_df, players_df, weekly_stats_df) -> list[ProspectFeatures]:
        """Pipeline: join data sources, build feature vectors, label outcomes, return list.
        - Join combine + draft_picks + players via ID map
        - Build feature vector per player-draft-year
        - For historical players (draft_year <= LATEST_CONFIRMABLE_YEAR): compute seasonal ranks, label outcome
        - For current class: outcome_bucket = None
        - Impute missing combine values with position-specific medians (Research pitfall #7)
        """
```
