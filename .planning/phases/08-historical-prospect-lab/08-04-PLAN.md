```python
"""Prospect model pipeline — CLI entry point.

Usage:
    python -m fantasy.prospects.pipeline [--positions QB,RB,WR,TE] [--current-class-year 2026]

Orchestrates:
1. Load data via NflReadPyLoader (combine, players, draft picks)
2. Load existing weekly stats from DuckDB (player_stats_weekly table, already ingested in Phase 1)
3. Build feature vectors via FeatureBuilder (historical + current class)
4. Label outcomes via HitClassifier (historical only — current class gets outcome_bucket=None)
5. Train position-specific models via ProspectModel (walk-forward validation)
6. Assign archetypes via ArchetypeClusterer (fit on historical, predict on all)
7. Find comps via CompFinder (for current class only)
8. Compute divergence flags via DivergenceEngine (for current class only)
9. Persist all outputs via ProspectRepo (upsert into 3 DuckDB tables)
"""
import argparse
import logging
from datetime import datetime, timezone

from fantasy.prospects.nflreadpy_loader import NflReadPyLoader
from fantasy.prospects.feature_builder import FeatureBuilder
from fantasy.prospects.hit_classifier import HitClassifier
from fantasy.prospects.prospect_model import ProspectModel
from fantasy.prospects.archetype_clusterer import ArchetypeClusterer
from fantasy.prospects.comp_finder import CompFinder
from fantasy.prospects.divergence_engine import DivergenceEngine
from fantasy.prospects.prospect_repo import ProspectRepo
from fantasy.prospects.models import ProspectModelOutput, SubFlag
from fantasy.prospects.constants import (
    HISTORICAL_START_YEAR, LATEST_CONFIRMABLE_YEAR, POSITION_FEATURES,
    MIN_COMP_COUNT, OUTCOME_HIT
)

logger = logging.getLogger(__name__)

POSITIONS = ["QB", "RB", "WR", "TE"]

def run_pipeline(
    db_path: str,
    positions: list[str] | None = None,
    current_class_year: int = 2026,
) -> dict:
    """Run the full prospect model pipeline.

    Args:
        db_path: Path to DuckDB database file
        positions: Positions to process (default: all 4)
        current_class_year: Draft year for the current class (default: 2026)

    Returns:
        dict with summary: {"positions_processed": [...], "prospects_scored": int, "comps_generated": int, "flags_generated": int}

    Pipeline steps (executed in order):
    1. NflReadPyLoader.load_combine(start_year=HISTORICAL_START_YEAR, end_year=current_class_year+1)
    2. NflReadPyLoader.load_players()
    3. NflReadPyLoader.load_draft_picks(start_year=HISTORICAL_START_YEAR, end_year=current_class_year+1)
    4. Read player_stats_weekly from DuckDB via SQL (already ingested in Phase 1)
    5. For each position:
       a. FeatureBuilder.build_all_features() — returns list[ProspectFeatures]
       b. Split into historical (draft_year <= LATEST_CONFIRMABLE_YEAR) and current (draft_year == current_class_year)
       c. HitClassifier.classify_cohort() — label outcomes on historical only
       d. ProspectModel(position).train() on historical with confirmed outcomes
       e. ProspectModel.walk_forward_validate() — log accuracy metrics
       f. ProspectModel.get_feature_importance() — log feature weights
       g. ArchetypeClusterer(position).fit() on historical pre-draft features
       h. ArchetypeClusterer.predict() on current class — assigns archetype_label
       i. ArchetypeClusterer.get_archetype_hit_rates() — computes hit-rate buckets
       j. CompFinder.find_comps() per current class prospect — 3 comps each
       k. DivergenceEngine.compute_divergence() per current class prospect — direction + magnitude
       l. DivergenceEngine.compute_sub_flags() per current class prospect — per-signal sub-flags
       m. Build ProspectModelOutput for each current class prospect
       n. ProspectRepo.upsert_features() — persist historical features
       o. ProspectRepo.upsert_model_outputs() — persist current class outputs
       p. ProspectRepo.upsert_sub_flags() — persist sub-flags
    6. Log summary and return
    """

def _determine_hit_rate_bucket(archetype_hit_rates: dict[str, dict[str, float]], archetype_label: str) -> str:
    """Convert archetype hit rate into bucket string for UI display.
    - hit_rate >= 0.40: "High hit rate"
    - 0.20 <= hit_rate < 0.40: "Moderate hit rate"
    - hit_rate < 0.20: "Low hit rate"
    Thresholds are intentionally simple — three buckets matching UI-SPEC badge colors.
    """

def _read_weekly_stats_from_db(db_path: str) -> "pl.DataFrame":
    """Read player_stats_weekly table from DuckDB.
    SELECT player_id, season, week, fantasy_points, position FROM player_stats_weekly
    WHERE position IN ('QB', 'RB', 'WR', 'TE')
    Returns Polars DataFrame.
    """

def _read_adp_baseline_from_db(db_path: str) -> dict[str, float]:
    """Read player_adp_baseline table from DuckDB.
    Returns dict of player_id -> adp float value.
    Used by DivergenceEngine for ADP divergence computation.
    """

if __name__ == "__main__":
    parser = argparse.ArgumentParser(description="Run prospect model pipeline")
    parser.add_argument("--db-path", required=True, help="Path to DuckDB database")
    parser.add_argument("--positions", default="QB,RB,WR,TE", help="Comma-separated positions")
    parser.add_argument("--current-class-year", type=int, default=2026, help="Current draft class year")
    args = parser.parse_args()
    result = run_pipeline(
        db_path=args.db_path,
        positions=args.positions.split(","),
        current_class_year=args.current_class_year,
    )
    print(f"Pipeline complete: {result}")
```

```python
from fastapi import APIRouter, Depends
from fantasy.routers.deps import get_db_connection  # existing DuckDB dependency
from fantasy.prospects.prospect_repo import ProspectRepo
from fantasy.prospects.models import ProspectModelOutput, HistoricalComp

router = APIRouter()

@router.get("/model-outputs/{league_id}", response_model=list[ProspectModelOutput])
async def get_model_outputs(league_id: str, conn=Depends(get_db_connection)):
    """Return all ProspectModelOutput objects for the current draft class.

    Per 08-UI-SPEC.md: this endpoint returns model outputs separately from Phase 7 rookie board data.
    Client joins by player_id. staleTime on client is 30 minutes (model outputs change infrequently).

    Note: league_id is accepted for API consistency with Phase 7's /rookie-board/{league_id}
    but prospect model outputs are not league-specific in Phase 8 — the same model outputs
    apply across leagues. The league_id parameter is reserved for future league-specific
    scoring adjustments. For now, return all current class model outputs regardless of league_id.
    """
    repo = ProspectRepo(conn)
    return repo.get_model_outputs(league_id)

@router.get("/comps/{player_id}", response_model=list[HistoricalComp])
async def get_comps(player_id: str, conn=Depends(get_db_connection)):
    """Return 3 HistoricalComp objects (ceiling/median/floor) for a single player.

    Per D-04: Three comps labeled by outcome role.
    Per D-06: Each comp has player_name, outcome_bucket, and one-line match_reason.
    """
    repo = ProspectRepo(conn)
    return repo.get_comps(player_id)
```
