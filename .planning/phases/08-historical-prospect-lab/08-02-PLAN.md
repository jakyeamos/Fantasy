**`hit_classifier.py`:**

```python
from fantasy.prospects.constants import (
    HIT_THRESHOLDS, MEDIOCRE_THRESHOLDS, MIN_OUTCOME_SEASONS,
    OUTCOME_HIT, OUTCOME_MEDIOCRE, OUTCOME_BUST
)

class HitClassifier:
    def classify_player(self, position: str, season_finish_ranks: list[int]) -> str | None:
        """Apply three-bucket classification per D-01, D-02.
        - Per D-03: timing NOT factored — counts qualifying seasons regardless of when
        - Returns None if fewer than MIN_OUTCOME_SEASONS available
        - Hit: finished top-N in M+ of first W seasons (position-specific)
        - Mediocre: met mediocre band but not hit threshold
        - Bust: below mediocre band
        """
        if len(season_finish_ranks) < MIN_OUTCOME_SEASONS:
            return None
        hit_cfg = HIT_THRESHOLDS[position]
        window = season_finish_ranks[:hit_cfg["window"]]
        qualifying_hit = sum(1 for r in window if r <= hit_cfg["top_n"])
        if qualifying_hit >= hit_cfg["min_seasons"]:
            return OUTCOME_HIT

        med_cfg = MEDIOCRE_THRESHOLDS[position]
        med_window = season_finish_ranks[:med_cfg["window"]]
        qualifying_med = sum(1 for r in med_window if r <= med_cfg["top_n"])
        if qualifying_med >= med_cfg["min_seasons"]:
            return OUTCOME_MEDIOCRE

        return OUTCOME_BUST

    def classify_cohort(self, features_with_ranks: list[dict]) -> list[dict]:
        """Batch classify a cohort. Each dict has position, season_finish_ranks keys."""
```

```python
from sklearn.ensemble import RandomForestClassifier
from sklearn.linear_model import LogisticRegression
from sklearn.inspection import permutation_importance
from sklearn.preprocessing import StandardScaler
from sklearn.impute import SimpleImputer

class ProspectModel:
    def __init__(self, position: str):
        self.position = position
        self.feature_names = POSITION_FEATURES[position]
        self.scaler = StandardScaler()
        self.imputer = SimpleImputer(strategy="median")
        # QB: LogisticRegression with strong L2 (small N, per research pitfall #6)
        # RB/WR/TE: RandomForest (more robust, handles nonlinear interactions)
        if position == "QB":
            self.model = LogisticRegression(C=0.1, max_iter=1000, random_state=42, multi_class="multinomial")
        else:
            self.model = RandomForestClassifier(n_estimators=200, max_features="sqrt", random_state=42)

    def walk_forward_validate(self, features_df: pl.DataFrame) -> dict:
        """Walk-forward validation per research Pattern 1.
        - Group by draft_year
        - Train on years [start..test_year-1], test on test_year
        - Start at index 3 for minimum training size
        - CRITICAL: fit scaler and imputer on train set only, transform test set
        - Returns dict with per-year accuracy and overall metrics
        """

    def train(self, X: np.ndarray, y: np.ndarray) -> None:
        """Fit imputer, scaler, and model on training data."""
        X_imputed = self.imputer.fit_transform(X)
        X_scaled = self.scaler.fit_transform(X_imputed)
        self.model.fit(X_scaled, y)

    def predict(self, X: np.ndarray) -> list[dict]:
        """Return probability distribution over hit/mediocre/bust for each row."""
        X_imputed = self.imputer.transform(X)
        X_scaled = self.scaler.transform(X_imputed)
        probs = self.model.predict_proba(X_scaled)
        return [{"hit": p[0], "mediocre": p[1], "bust": p[2]} for p in probs]

    def get_feature_importance(self, X_val: np.ndarray, y_val: np.ndarray) -> dict[str, float]:
        """Permutation importance (NOT impurity-based) per research recommendation.
        Uses sklearn.inspection.permutation_importance with n_repeats=30.
        Returns {feature_name: importance_score}.
        """
        X_imputed = self.imputer.transform(X_val)
        X_scaled = self.scaler.transform(X_imputed)
        result = permutation_importance(self.model, X_scaled, y_val, n_repeats=30, random_state=42)
        return {name: float(imp) for name, imp in zip(self.feature_names, result.importances_mean)}
```

**`archetype_clusterer.py`:**

```python
from sklearn.cluster import KMeans
from sklearn.preprocessing import StandardScaler
import numpy as np

# Archetype counts per position (Claude's Discretion)
ARCHETYPE_K = {"QB": 3, "RB": 4, "WR": 4, "TE": 3}

# Archetype label vocabulary (Claude's Discretion — named after inspecting centroids)
# These are initial labels; renamed after first model run based on centroid characteristics
ARCHETYPE_LABELS = {
    "QB": ["Dual-Threat", "Pocket Passer", "Project QB"],
    "RB": ["Speed Back", "Power Back", "Receiving Back", "Versatile Back"],
    "WR": ["Deep Threat", "Slot Technician", "Contested Catch", "YAC Monster"],
    "TE": ["Receiving TE", "Blocking TE", "Hybrid TE"],
}

# Clustering features: pre-draft only, NO outcome features (Research anti-pattern warning)
CLUSTER_FEATURES = ["age_at_draft", "draft_ovr", "forty", "weight", "height", "vertical"]

class ArchetypeClusterer:
    def __init__(self, position: str):
        self.position = position
        self.k = ARCHETYPE_K[position]
        self.labels = ARCHETYPE_LABELS[position]
        self.scaler = StandardScaler()
        self.kmeans = KMeans(n_clusters=self.k, n_init=10, random_state=42)
        self.fitted = False

    def fit(self, X: np.ndarray) -> None:
        """Fit KMeans on pre-draft features ONLY.
        X columns must match CLUSTER_FEATURES — no outcome_bucket, no NFL stats.
        """
        X_scaled = self.scaler.fit_transform(X)
        self.kmeans.fit(X_scaled)
        self.fitted = True

    def predict(self, X: np.ndarray) -> list[str]:
        """Assign archetype label to each prospect.
        Returns list of archetype label strings from ARCHETYPE_LABELS[position].
        """
        X_scaled = self.scaler.transform(X)
        cluster_ids = self.kmeans.predict(X_scaled)
        return [self.labels[cid] for cid in cluster_ids]

    def get_archetype_hit_rates(self, X: np.ndarray, outcomes: list[str]) -> dict[str, dict[str, float]]:
        """For each archetype cluster, compute hit/mediocre/bust distribution.
        Returns {"Dual-Threat": {"hit": 0.3, "mediocre": 0.4, "bust": 0.3}, ...}.
        Used to derive hit-rate buckets per archetype per PROS-03.
        """

    def get_centroids(self) -> dict[str, list[float]]:
        """Return cluster centroids for storage in constants.py or model cache."""
        centroids = self.scaler.inverse_transform(self.kmeans.cluster_centers_)
        return {label: centroids[i].tolist() for i, label in enumerate(self.labels)}
```
