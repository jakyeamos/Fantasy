**`frontend/src/api/prospects.ts`:** TanStack Query hooks:

```typescript
import { useQuery } from "@tanstack/react-query";
import type { ProspectModelOutput, HistoricalComp } from "../types/prospects";

const API_BASE = import.meta.env.VITE_API_URL ?? "";

export function useProspectModelOutputs(leagueId: string) {
  return useQuery<ProspectModelOutput[]>({
    queryKey: ["prospect-model-outputs", leagueId],
    queryFn: async () => {
      const res = await fetch(`${API_BASE}/prospects/model-outputs/${leagueId}`);
      if (!res.ok) throw new Error("Failed to fetch prospect model outputs");
      return res.json();
    },
    staleTime: 30 * 60 * 1000,  // 30 minutes — per UI-SPEC: model outputs change less frequently
    enabled: !!leagueId,
  });
}

export function useProspectComps(playerId: string) {
  return useQuery<HistoricalComp[]>({
    queryKey: ["prospect-comps", playerId],
    queryFn: async () => {
      const res = await fetch(`${API_BASE}/prospects/comps/${playerId}`);
      if (!res.ok) throw new Error("Failed to fetch prospect comps");
      return res.json();
    },
    staleTime: 30 * 60 * 1000,
    enabled: !!playerId,
  });
}
```
