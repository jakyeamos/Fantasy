```python
from scipy.spatial.distance import cdist
from sklearn.preprocessing import StandardScaler
import numpy as np
from fantasy.prospects.constants import (
    DRAFT_CAPITAL_TIERS, MIN_COMP_COUNT, CLUSTER_FEATURES,
    OUTCOME_HIT, OUTCOME_MEDIOCRE, OUTCOME_BUST
)
from fantasy.prospects.models import HistoricalComp, ProspectFeatures

class CompFinder:
    def __init__(self):
        self.scaler = StandardScaler()

    def get_draft_capital_tier(self, draft_ovr: int) -> str:
        """Bucket NFL draft overall pick into a tier per DRAFT_CAPITAL_TIERS.
        DRAFT_CAPITAL_TIERS = {"Tier 1": (1, 12), "Tier 2": (13, 36), "Day 2": (37, 72), "Day 3": (73, 255)}
        Returns the tier name string. Picks above 255 return "Day 3".
        """
        for tier_name, (lo, hi) in DRAFT_CAPITAL_TIERS.items():
            if lo <= draft_ovr <= hi:
                return tier_name
        return "Day 3"  # fallback for undrafted/late picks

    def find_comps(
        self,
        prospect: ProspectFeatures,
        historical_pool: list[ProspectFeatures],
        feature_names: list[str] | None = None,
    ) -> tuple[list[HistoricalComp], bool]:
        """Find ceiling/median/floor comps for a single prospect per D-04, D-05, D-06.

        Algorithm (per research Pattern 4):
        1. Compute prospect's draft capital tier via get_draft_capital_tier(prospect.draft_ovr)
        2. Filter historical_pool to same position AND same tier
        3. If filtered pool size < MIN_COMP_COUNT: set low_confidence = True
        4. Extract feature matrix using CLUSTER_FEATURES (or feature_names if provided)
           - Use the same feature subset as archetype clustering (pre-draft only, no outcome features)
        5. Fit StandardScaler on the filtered pool, transform both pool and prospect
        6. Compute Euclidean distance from prospect to each candidate using scipy.spatial.distance.cdist
        7. Partition candidates by outcome_bucket (hit / mediocre / bust)
        8. From each partition, select the nearest neighbor by distance
        9. Generate match_reason: identify top 2-3 features with largest absolute distance contribution
           - Format: "similar {feature1}, {feature2}, and {feature3} profile"
           - Map feature names to human-readable: age_at_draft -> "age", draft_ovr -> "draft capital",
             forty -> "speed", weight -> "size", height -> "size", vertical -> "athleticism",
             college_rec_ypg -> "college production", college_mkt_share_proxy -> "market share"
        10. Return (list[HistoricalComp], low_confidence)
            - List has up to 3 comps. If an outcome bucket has no candidates, that slot is omitted (not None).

        Per D-05: comps are relative to ADP tier — "ceiling" means best outcome at this draft capital level.
        Per D-06: each comp has player_name, outcome_bucket, and one-line match_reason. No peak stats.
        """

    def _compute_feature_contributions(
        self,
        prospect_vec: np.ndarray,
        comp_vec: np.ndarray,
        feature_names: list[str]
    ) -> list[str]:
        """Return top 2-3 feature names that contributed most to the distance.
        Computes per-feature absolute difference, sorts descending, returns top 2-3 names.
        """
```

```python
import numpy as np
from fantasy.prospects.constants import MIN_COMP_COUNT
from fantasy.prospects.models import SubFlag, ProspectFeatures

class DivergenceEngine:
    """Computes over/undervalue flags per PROS-05 and decisions D-08 through D-11.

    Core methodology (per research Pattern 5):
    - Compare prospect's current dynasty ADP to historical median ADP of players
      with same archetype + draft capital tier who achieved "hit" outcomes
    - Divergence magnitude = current_adp - historical_median_adp_for_hits
      - Positive magnitude = overvalued (paying more than precedent)
      - Negative magnitude = undervalued (paying less than precedent)
    - Per D-08: direction + magnitude stated as "N spots overvalued/undervalued"
    - Per D-11: descriptive only — no action recommendation
    """

    def compute_divergence(
        self,
        prospect: ProspectFeatures,
        prospect_adp: float,
        historical_pool: list[ProspectFeatures],
        historical_adps: dict[str, float],  # player_id -> dynasty ADP
        archetype_label: str,
        draft_capital_tier: str,
    ) -> tuple[str | None, int | None, bool]:
        """Compute net over/undervalue verdict.

        Steps:
        1. Filter historical_pool to same position + same draft_capital_tier + same archetype_label
        2. If filtered count < MIN_COMP_COUNT: return (None, None, True) — low confidence per D-10
        3. From filtered pool, select those with outcome_bucket == "hit"
        4. If no hits exist: return (None, 0, low_confidence) — cannot compute divergence without hit outcomes
        5. Compute median dynasty ADP of hit-outcome players from historical_adps
        6. magnitude = round(prospect_adp - median_hit_adp)
           - In dynasty ADP, LOWER number = MORE valuable (pick 1 > pick 50)
           - So if prospect_adp < median_hit_adp: prospect is going EARLIER than hits historically went = overvalued
           - If prospect_adp > median_hit_adp: prospect is going LATER = undervalued
        7. Direction:
           - magnitude < 0 (prospect going earlier): direction = "overvalued", display_magnitude = abs(magnitude)
           - magnitude > 0 (prospect going later): direction = "undervalued", display_magnitude = magnitude
           - magnitude == 0: direction = None (no divergence)
        8. Return (direction, display_magnitude, low_confidence)
        """

    def compute_sub_flags(
        self,
        prospect: ProspectFeatures,
        archetype_medians: dict[str, float],  # feature_name -> archetype cluster median value
        archetype_stds: dict[str, float],      # feature_name -> archetype cluster std dev
    ) -> list[SubFlag]:
        """Compute per-signal sub-flags per D-09.

        For each signal category, compute z-score of prospect vs archetype cluster median:
        - "Draft Capital": z-score of draft_ovr (lower draft_ovr = positive signal)
        - "Age": z-score of age_at_draft (lower age = positive signal for most positions)
        - "Production": z-score of college production proxy (higher = positive)
        - "Athletic Testing": z-score of forty, vertical, etc. (composite or primary metric)
        - "Market Share": z-score of college_mkt_share_proxy (higher = positive)

        Direction assignment:
        - |z-score| <= 0.5: direction = "neutral"
        - z-score favorable (positive for production, negative for age): direction = "positive"
        - z-score unfavorable: direction = "negative"

        magnitude_str format examples (rendered verbatim by frontend per D-09):
        - "0.5 years older than median — slight negative"
        - "top-10% market share — strong positive"
        - "within normal range"
        - "2.3 years younger — strong positive"
        - "drafted 15 spots later — moderate negative"

        Signal-to-feature mapping:
        - "Draft Capital" -> draft_ovr (lower is better, so negative z = positive signal)
        - "Age" -> age_at_draft (lower is better, so negative z = positive signal)
        - "Production" -> college_rec_ypg or college_rush_ypg depending on position (higher is better)
        - "Athletic Testing" -> forty (lower is better for forty — flip sign)
        - "Market Share" -> college_mkt_share_proxy (higher is better)

        Return list of SubFlag objects. Each has signal_name, direction, magnitude_str.
        If a feature value is None (missing data), skip that sub-flag — do not include it.
        """

    def _format_magnitude_str(
        self,
        signal_name: str,
        raw_diff: float,
        z_score: float,
        direction: str,
    ) -> str:
        """Format human-readable magnitude string.
        - Strength label: |z| > 1.5 = "strong", 0.5 < |z| <= 1.5 = "moderate", |z| <= 0.5 = "within normal range"
        - Unit label: age -> "years", draft_ovr -> "spots", forty -> "seconds", others -> raw value
        """
```

- Per D-11: No action text. The engine outputs direction + magnitude only. "8 spots overvalued" is the full output.
- Dynasty ADP convention: lower ADP number = more valuable pick. ADP 1 = most valuable.
