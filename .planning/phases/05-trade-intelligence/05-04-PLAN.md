
"See reroutes" handler — calls evaluate mutation again with `include_reroutes: true`:
```typescript
const handleSeeReroutes = async () => {
  setRerouteLoading(true);
  setRerouteSheetOpen(true);
  try {
    const result = await evaluateMutation.mutateAsync({
      ...currentRequest,
      include_reroutes: true,
      include_package: false,
    });
    setRerouteData(result.reroutes ?? null);
  } finally {
    setRerouteLoading(false);
  }
};
```

"Build Package" handler — calls evaluate mutation with `include_package: true`:
```typescript
const handleBuildPackage = async () => {
  setPackageLoading(true);
  try {
    const result = await evaluateMutation.mutateAsync({
      ...currentRequest,
      include_reroutes: false,
      include_package: true,
    });
    setPackageData(result.package ?? null);
  } finally {
    setPackageLoading(false);
  }
};
```

```tsx
<RerouteSheet

    package={packageData}
    isLoading={packageLoading}
  />
```
