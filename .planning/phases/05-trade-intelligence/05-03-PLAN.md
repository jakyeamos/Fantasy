export type AssetType = "player" | "pick";
export type ConfidenceLevel = "HIGH" | "MEDIUM" | "LOW";

  asset_type: AssetType;

  pick_owner_roster_id?: number;
  pick_owner_name?: string;   // for display

  roster_id: number;

  league_id: string;

**Step 3: Create `frontend/src/hooks/useTradeMutation.ts`:**

```typescript
import { useMutation, useQuery } from "@tanstack/react-query";
import type { TradeRequest, TradeEvaluation, PlayerSearchResult, PickSearchResult } from "@/types/trade";

const API_BASE = import.meta.env.VITE_API_URL ?? "";

export function useTradeEvaluation() {
  return useMutation({
    mutationFn: async (request: TradeRequest): Promise<TradeEvaluation> => {
      const res = await fetch(`${API_BASE}/trade/evaluate`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(request),
      });
      if (!res.ok) throw new Error("Evaluation failed");
      return res.json();
    },
  });
}

export function usePlayerSearch(leagueId: string, query: string, rosterId?: number) {
  return useQuery({
    queryKey: ["player-search", leagueId, query, rosterId],
    queryFn: async (): Promise<PlayerSearchResult[]> => {
      const params = new URLSearchParams({ league_id: leagueId, q: query });
      if (rosterId != null) params.set("roster_id", String(rosterId));
      const res = await fetch(`${API_BASE}/trade/players/search?${params}`);
      if (!res.ok) throw new Error("Player search failed");
      return res.json();
    },
    enabled: query.length >= 1,
    staleTime: 30_000,
  });
}

export function usePickSearch(leagueId: string, rosterId?: number) {
  return useQuery({
    queryKey: ["pick-search", leagueId, rosterId],
    queryFn: async (): Promise<PickSearchResult[]> => {
      const params = new URLSearchParams({ league_id: leagueId });
      if (rosterId != null) params.set("roster_id", String(rosterId));
      const res = await fetch(`${API_BASE}/trade/picks/search?${params}`);
      if (!res.ok) throw new Error("Pick search failed");
      return res.json();
    },
    staleTime: 60_000,
  });
}
```
