```typescript
export async function saveDraftOrderRule(
  leagueId: string,
  rule: LeagueDraftOrderRule,
): Promise<DraftOrderRuleResponse> {
  const response = await fetch(`/api/picks/${leagueId}/draft-order-rule`, {
    method: "PUT",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(rule),
  })
  if (!response.ok) {
    throw new Error("Failed to save draft order rule")
  }
  return (await response.json()) as DraftOrderRuleResponse
}
```

Create the `DraftOrderRuleForm` component inside the file (not a separate file — it is specific to this route). Per D-09: editor lives in league settings. Per D-10: single-page form with all 3 fields visible.

```tsx
function DraftOrderRuleForm({ leagueId }: { leagueId: string }) {
  const queryClient = useQueryClient()
  const ruleQuery = useQuery(draftOrderRuleOptions(leagueId))
  const [editing, setEditing] = useState(false)
  const [basis, setBasis] = useState<string>("")
  const [playoffOrdering, setPlayoffOrdering] = useState<string>("")
  const [tiebreaker, setTiebreaker] = useState<string>("")
  const [saveFlash, setSaveFlash] = useState(false)

  // Sync form state when query loads
  const rule = ruleQuery.data?.rule ?? null
  const isConfigured = rule !== null && !editing

  // Initialize form state from loaded rule
  useEffect(() => {
    if (rule && !editing) {
      setBasis(rule.non_playoff_basis)
      setPlayoffOrdering(rule.playoff_ordering)
      setTiebreaker(rule.tiebreaker)
    }
  }, [rule, editing])

  const mutation = useMutation({
    mutationFn: (newRule: LeagueDraftOrderRule) => saveDraftOrderRule(leagueId, newRule),
    onSuccess: async () => {
      await queryClient.invalidateQueries({ queryKey: ["picks", leagueId] })
      setSaveFlash(true)
      setEditing(false)
      setTimeout(() => setSaveFlash(false), 1500)
    },
  })

  const allFieldsFilled = basis !== "" && playoffOrdering !== "" && tiebreaker !== ""

  function handleSubmit(e: React.FormEvent) {
    e.preventDefault()
    if (!allFieldsFilled) return
    mutation.mutate({
      non_playoff_basis: basis as LeagueDraftOrderRule["non_playoff_basis"],
      playoff_ordering: playoffOrdering as LeagueDraftOrderRule["playoff_ordering"],
      tiebreaker: tiebreaker as LeagueDraftOrderRule["tiebreaker"],
    })
  }

  // Display-only mode when configured and not editing
  if (isConfigured) {
    return (
      <div id="draft-order-rule">
        <Card>
          <CardHeader className="flex flex-row items-center justify-between">
            <CardTitle className="text-lg">Draft Order Rule</CardTitle>
            <button
              type="button"
              className="text-sm text-primary underline cursor-pointer"
              onClick={() => setEditing(true)}
            >
              Edit
            </button>
          </CardHeader>
          <CardContent className="space-y-2">
            <p className="text-sm"><span className="text-muted-foreground">Non-playoff ordering:</span> {basis === "inverse_standings" ? "Inverse standings" : "Max points for"}</p>
            <p className="text-sm"><span className="text-muted-foreground">Playoff team ordering:</span> {playoffOrdering === "by_finish" ? "By playoff finish" : playoffOrdering === "by_record" ? "By regular season record" : "By regular season points for"}</p>
            <p className="text-sm"><span className="text-muted-foreground">Tiebreaker:</span> {tiebreaker === "points_against" ? "Points against" : tiebreaker === "points_for" ? "Points for" : "Commissioner discretion"}</p>
          </CardContent>
        </Card>
      </div>
    )
  }

  // Edit mode (or never configured)
  return (
    <div id="draft-order-rule">
      <Card>
        <CardHeader>
          <CardTitle className="text-lg">Draft Order Rule</CardTitle>
          <p className="text-sm text-muted-foreground">Configure how this league's draft order is determined.</p>
        </CardHeader>
        <CardContent>
          <form onSubmit={handleSubmit} className="space-y-6">
            {/* Field 1: Non-playoff ordering — RadioGroup per UI-SPEC note 9 */}
            <div className="space-y-2">
              <Label className="text-sm">Non-playoff ordering</Label>
              <RadioGroup value={basis} onValueChange={setBasis}>
                <div className="flex items-center gap-2">
                  <RadioGroupItem value="inverse_standings" id="basis-inverse" />
                  <Label htmlFor="basis-inverse" className="text-sm font-normal">Inverse standings</Label>
                </div>
                <div className="flex items-center gap-2">
                  <RadioGroupItem value="max_points_for" id="basis-maxpf" />
                  <Label htmlFor="basis-maxpf" className="text-sm font-normal">Max points for</Label>
                </div>
              </RadioGroup>
              <p className="text-xs text-muted-foreground">How are non-playoff teams ordered at draft time?</p>
            </div>

            {/* Field 2: Playoff team ordering — Select per UI-SPEC note 9 */}
            <div className="space-y-2">
              <Label className="text-sm">Playoff team ordering</Label>
              <Select value={playoffOrdering} onValueChange={setPlayoffOrdering}>
                <SelectTrigger className="w-full">
                  <SelectValue placeholder="Select playoff ordering" />
                </SelectTrigger>
                <SelectContent>
                  <SelectItem value="by_finish">Playoff teams by finish</SelectItem>
                  <SelectItem value="by_record">Playoff teams by record</SelectItem>
                  <SelectItem value="by_points_for">Playoff teams by points for</SelectItem>
                </SelectContent>
              </Select>
              <p className="text-xs text-muted-foreground">How are teams that made the playoffs slotted at the end of the draft order?</p>
            </div>

            {/* Field 3: Tiebreaker — Select per UI-SPEC note 9 */}
            <div className="space-y-2">
              <Label className="text-sm">Tiebreaker</Label>
              <Select value={tiebreaker} onValueChange={setTiebreaker}>
                <SelectTrigger className="w-full">
                  <SelectValue placeholder="Select tiebreaker" />
                </SelectTrigger>
                <SelectContent>
                  <SelectItem value="points_against">Points against (higher = worse pick)</SelectItem>
                  <SelectItem value="points_for">Points for (lower = better pick)</SelectItem>
                  <SelectItem value="commissioner">Commissioner discretion</SelectItem>
                </SelectContent>
              </Select>
              <p className="text-xs text-muted-foreground">How are tied teams in the non-playoff ordering separated?</p>
            </div>

            {/* Save row */}
            <div className="space-y-2">
              <Button
                type="submit"
                size="sm"
                disabled={!allFieldsFilled || mutation.isPending}
                className={saveFlash ? "text-green-600" : ""}
              >
                {mutation.isPending ? "Saving..." : saveFlash ? "Saved" : "Save Rule"}
              </Button>
              {!allFieldsFilled && (
                <p className="text-xs text-muted-foreground">All three fields are required before pick projections unlock.</p>
              )}
              {mutation.isError && (
                <p className="text-xs text-destructive">Save failed. Try again.</p>
              )}
            </div>
          </form>
        </CardContent>
      </Card>
    </div>
  )
}
```
