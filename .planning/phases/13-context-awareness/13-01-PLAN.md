# backend/src/fantasy/context/context_repo.py
from __future__ import annotations

from dataclasses import dataclass
from datetime import datetime, timezone
from typing import Optional

import duckdb


@dataclass
class FreshnessRow:
    domain: str
    last_updated: datetime | None
    notes: str | None


class ContextRepo:
    def __init__(self, conn: duckdb.DuckDBPyConnection) -> None:
        self._conn = conn

    # --- Calendar override ---

    def upsert_override(
        self,
        league_id: str,
        state: str,
        expires_at: datetime | None = None,
        set_by: str = "user",
    ) -> None:
        self._conn.execute(
            """
            INSERT INTO calendar_overrides (league_id, state, set_by, set_at, expires_at)
            VALUES (?, ?, ?, ?, ?)
            ON CONFLICT (league_id) DO UPDATE SET
                state = excluded.state,
                set_by = excluded.set_by,
                set_at = excluded.set_at,
                expires_at = excluded.expires_at
            """,
            [league_id, state, set_by, datetime.now(tz=timezone.utc), expires_at],
        )

    def get_override(self, league_id: str) -> str | None:
        """Returns the active state string if a non-expired override exists, else None."""
        now = datetime.now(tz=timezone.utc)
        row = self._conn.execute(
            """
            SELECT state, expires_at FROM calendar_overrides
            WHERE league_id = ?
            """,
            [league_id],
        ).fetchone()
        if row is None:
            return None
        state, expires_at = row
        if expires_at is not None and expires_at < now:
            return None
        return state

    def clear_override(self, league_id: str) -> None:
        self._conn.execute(
            "DELETE FROM calendar_overrides WHERE league_id = ?", [league_id]
        )

    # --- Freshness domains ---

    def get_freshness_rows(
        self, league_id: str, domains: list[str]
    ) -> dict[str, FreshnessRow]:
        if not domains:
            return {}
        placeholders = ", ".join(["?" for _ in domains])
        rows = self._conn.execute(
            f"""
            SELECT domain, last_updated, notes FROM freshness_domains
            WHERE league_id = ? AND domain IN ({placeholders})
            """,
            [league_id, *domains],
        ).fetchall()
        return {
            row[0]: FreshnessRow(domain=row[0], last_updated=row[1], notes=row[2])
            for row in rows
        }

    def upsert_freshness(
        self,
        league_id: str,
        domain: str,
        last_updated: datetime,
        notes: str | None = None,
    ) -> None:
        self._conn.execute(
            """
            INSERT INTO freshness_domains (league_id, domain, last_updated, notes)
            VALUES (?, ?, ?, ?)
            ON CONFLICT (league_id, domain) DO UPDATE SET
                last_updated = excluded.last_updated,
                notes = excluded.notes
            """,
            [league_id, domain, last_updated, notes],
        )
