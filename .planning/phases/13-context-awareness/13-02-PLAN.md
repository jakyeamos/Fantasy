# backend/src/fantasy/context/calendar_service.py
from __future__ import annotations

from datetime import datetime, timezone
from typing import Callable

from fantasy.context.constants import (
    CALENDAR_GUIDANCE,
    CALENDAR_WINDOWS,
    FALLBACK_STATE,
    CalendarState,
)
from fantasy.context.context_repo import ContextRepo
from fantasy.context.models import CalendarContext

ClockFn = Callable[[], datetime]


def _in_range(m: int, d: int, sm: int, sd: int, em: int, ed: int) -> bool:
    """Check if (month m, day d) falls within [sm/sd, em/ed] inclusive.
    Handles cross-year ranges (e.g., Nov 22 – Jan 12) when sm > em."""
    if sm <= em:
        if sm < m < em:
            return True
        if m == sm and d >= sd:
            return True
        if m == em and d <= ed:
            return True
    else:
        # Cross-year range (e.g., Nov 22 – Jan 12)
        if m > sm or (m == sm and d >= sd):
            return True
        if m < em or (m == em and d <= ed):
            return True
    return False


def _detect_state(now: datetime) -> str:
    month, day = now.month, now.day
    for start_m, start_d, end_m, end_d, state in CALENDAR_WINDOWS:
        if _in_range(month, day, start_m, start_d, end_m, end_d):
            return state
    return FALLBACK_STATE


class CalendarService:
    def __init__(
        self,
        repo: ContextRepo,
        now: ClockFn = lambda: datetime.now(tz=timezone.utc),
    ) -> None:
        self._repo = repo
        self._now = now

    def active_state(self, league_id: str) -> str:
        override = self._repo.get_override(league_id)
        if override is not None:
            return override
        return _detect_state(self._now())

    def get_context(self, league_id: str) -> CalendarContext:
        override = self._repo.get_override(league_id)
        is_override = override is not None
        state = override if is_override else _detect_state(self._now())
        note = CALENDAR_GUIDANCE.get((state, "general"))
        return CalendarContext(
            active_state=state,
            is_override=is_override,
            override_set_by="user" if is_override else None,
            detected_at=self._now(),
        )

# backend/src/fantasy/context/freshness_service.py
from __future__ import annotations

from datetime import datetime, timedelta, timezone
from typing import Callable

from fantasy.context.constants import FRESHNESS_THRESHOLDS
from fantasy.context.context_repo import ContextRepo
from fantasy.context.models import FreshnessTag

ClockFn = Callable[[], datetime]


class FreshnessService:
    def __init__(
        self,
        repo: ContextRepo,
        now: ClockFn = lambda: datetime.now(tz=timezone.utc),
    ) -> None:
        self._repo = repo
        self._now = now

    def get_tags(self, league_id: str, domains: list[str]) -> list[FreshnessTag]:
        rows = self._repo.get_freshness_rows(league_id, domains)
        now = self._now()
        tags: list[FreshnessTag] = []
        for domain in domains:
            row = rows.get(domain)
            threshold_hours = FRESHNESS_THRESHOLDS.get(domain, 48)
            if row is None or row.last_updated is None:
                tags.append(
                    FreshnessTag(
                        domain=domain,
                        last_updated=None,
                        is_stale=True,
                        warning=(
                            f"{domain.replace('_', ' ').title()} data has never been updated"
                            " — verify before acting"
                        ),
                    )
                )
                continue
            lu = row.last_updated
            if lu.tzinfo is None:
                lu = lu.replace(tzinfo=timezone.utc)
            age = now - lu
            is_stale = age > timedelta(hours=threshold_hours)
            warning: str | None = None
            if is_stale:
                hours_old = int(age.total_seconds() / 3600)
                warning = (
                    f"{domain.replace('_', ' ').title()} data is {hours_old}h old"
                    " — verify before acting"
                )
            tags.append(
                FreshnessTag(
                    domain=domain,
                    last_updated=row.last_updated,
                    is_stale=is_stale,
                    warning=warning,
                )
            )
        return tags

    def mark_refreshed(self, league_id: str, domain: str, notes: str | None = None) -> None:
        self._repo.upsert_freshness(league_id, domain, self._now(), notes=notes)
```

from fantasy.context.constants import FRESHNESS_THRESHOLDS, OVERRIDE_TTL_DAYS
from fantasy.context.context_repo import ContextRepo
from fantasy.context.freshness_service import FreshnessService
from fantasy.context.models import CalendarContext, FreshnessTag
from fantasy.routers.deps import get_read_db_conn, get_write_db_conn

router = APIRouter(prefix="/context", tags=["context"])


# ── Calendar state ─────────────────────────────────────────────────────────────

@router.get("/{league_id}/calendar", response_model=CalendarContext)
def get_calendar_context(
    league_id: str,
    conn: duckdb.DuckDBPyConnection = Depends(get_read_db_conn),
) -> CalendarContext:
    repo = ContextRepo(conn)
    service = CalendarService(repo=repo)
    return service.get_context(league_id)


class CalendarOverrideRequest(BaseModel):
    state: str  # CalendarState literal
    ttl_days: int | None = None  # defaults to OVERRIDE_TTL_DAYS


@router.post("/{league_id}/calendar/override", response_model=CalendarContext)
def set_calendar_override(
    league_id: str,
    body: CalendarOverrideRequest,
    conn: duckdb.DuckDBPyConnection = Depends(get_write_db_conn),
) -> CalendarContext:
    ttl = body.ttl_days if body.ttl_days is not None else OVERRIDE_TTL_DAYS
    expires_at = datetime.now(tz=timezone.utc) + timedelta(days=ttl)
    repo = ContextRepo(conn)
    repo.upsert_override(league_id, body.state, expires_at=expires_at)
    service = CalendarService(repo=repo)
    return service.get_context(league_id)


@router.delete("/{league_id}/calendar/override", response_model=CalendarContext)
def clear_calendar_override(
    league_id: str,
    conn: duckdb.DuckDBPyConnection = Depends(get_write_db_conn),
) -> CalendarContext:
    repo = ContextRepo(conn)
    repo.clear_override(league_id)
    service = CalendarService(repo=repo)
    return service.get_context(league_id)


# ── Freshness domains ──────────────────────────────────────────────────────────

@router.get("/{league_id}/freshness", response_model=list[FreshnessTag])
def get_freshness(
    league_id: str,
    conn: duckdb.DuckDBPyConnection = Depends(get_read_db_conn),
) -> list[FreshnessTag]:
    repo = ContextRepo(conn)
    service = FreshnessService(repo=repo)
    domains = list(FRESHNESS_THRESHOLDS.keys())
    return service.get_tags(league_id, domains)


class FreshnessRefreshRequest(BaseModel):
    domain: str
    notes: str | None = None


@router.post("/{league_id}/freshness/refresh", response_model=FreshnessTag)
def mark_domain_refreshed(
    league_id: str,
    body: FreshnessRefreshRequest,
    conn: duckdb.DuckDBPyConnection = Depends(get_write_db_conn),
) -> FreshnessTag:
    if body.domain not in FRESHNESS_THRESHOLDS:
        raise HTTPException(
            status_code=422,
            detail=f"Unknown domain '{body.domain}'. Valid domains: {list(FRESHNESS_THRESHOLDS.keys())}",
        )
    repo = ContextRepo(conn)
    service = FreshnessService(repo=repo)
    service.mark_refreshed(league_id, body.domain, notes=body.notes)
    tags = service.get_tags(league_id, [body.domain])
    return tags[0]
```
