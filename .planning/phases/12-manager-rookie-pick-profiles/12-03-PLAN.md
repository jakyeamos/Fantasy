```python
def _generate_picks_buyer_reroute(
    self,
    league_id: str,
    counterparty_roster_id: int,
    user_receive_assets: list,  # assets user would receive in proposed trade
) -> RerouteResult | None:
    """Fire when: counterparty has pick_premium_score >= threshold AND user is not receiving picks.
    Silent omission when evidence is thin (D-17) — only fire when score is non-None (D-17 requires
    None = no evidence = silent omit).
    """
    # Check if user is already receiving picks in this trade
    user_receiving_picks = any(
        getattr(a, "asset_type", None) == "pick" for a in user_receive_assets
    )
    if user_receiving_picks:
        return None

    rp_repo = RookiePickDataRepo(self._conn)
    rp = rp_repo.get_profile(league_id, counterparty_roster_id)
    if rp is None:
        return None
    # D-17: None = no evidence = silent omit. 0.0 = evidence, no premium = also omit (below threshold)
    if rp.pick_premium_score is None or rp.pick_premium_score < PICK_PREMIUM_THRESHOLD:
        return None
    # Sufficient evidence + above threshold — emit reroute
    return RerouteResult(
        reroute_type="picks_buyer",
        headline=f"Consider including a pick — this manager consistently pays a premium for draft capital",
        reasoning=(
            f"This manager has a pick-premium score of {rp.pick_premium_score:.2f} based on "
            f"{rp.pick_trade_evidence} pick-related trade(s). "
            "Adding a pick to the package structure may improve acceptance odds."
        ),
        suggested_assets=None,
    )
```
