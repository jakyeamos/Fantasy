```python
from pydantic import BaseModel, ConfigDict

class RookiePickProfile(BaseModel):
    model_config = ConfigDict(frozen=False)

    league_id: str
    roster_id: int
    computed_at: str
    pick_premium_score: float | None = None
    """None = insufficient pick-trade evidence. 0.0 = evidence exists, no premium detected."""
    pick_trade_evidence: int = 0
    draft_selection_count: int = 0
    positional_tendency: dict[str, float] = {}
    """Per-position deviation from league average positional share (D-13).
    E.g. {"WR": 0.12, "RB": -0.08} means WR-heavy relative to league."""
    dominant_archetype: str | None = None
    """Most frequent archetype bet across draft selections (min 2 of same label). (D-14)"""
    archetype_pattern: dict[str, int] = {}
    """Archetype label -> selection count across all drafts."""
    show_draft_picks_tab: bool = False
    """True when pick_trade_evidence + draft_selection_count >= MIN_ROOKIE_PICK_EVIDENCE (D-02)."""
    draft_selection_history: list[dict] = []
    """Per-season draft selections list for Draft & Picks tab table."""
```

```python
from __future__ import annotations

import json
from typing import Any

import duckdb

from fantasy.rookie_pick.models import RookiePickProfile


class RookiePickRepo:
    def __init__(self, conn: duckdb.DuckDBPyConnection) -> None:
        self._conn = conn

    def get_profile(self, league_id: str, roster_id: int) -> RookiePickProfile | None:
        row = self._conn.execute(
            """
            SELECT league_id, roster_id, computed_at, pick_premium_score,
                   pick_trade_evidence, draft_selection_count,
                   positional_tendency_json, dominant_archetype,
                   archetype_pattern_json, show_draft_picks_tab
            FROM manager_rookie_pick_profiles
            WHERE league_id = ? AND roster_id = ?
            """,
            [league_id, roster_id],
        ).fetchone()
        if row is None:
            return None
        (lid, rid, cat, pps, pte, dsc, ptj, da, apj, sdt) = row
        return RookiePickProfile(
            league_id=lid,
            roster_id=rid,
            computed_at=str(cat),
            pick_premium_score=pps,
            pick_trade_evidence=pte,
            draft_selection_count=dsc,
            positional_tendency=json.loads(ptj or "{}"),
            dominant_archetype=da,
            archetype_pattern=json.loads(apj or "{}"),
            show_draft_picks_tab=bool(sdt),
        )

    def upsert_profile(self, profile: RookiePickProfile) -> None:
        existing_id = self._conn.execute(
            "SELECT id FROM manager_rookie_pick_profiles WHERE league_id = ? AND roster_id = ?",
            [profile.league_id, profile.roster_id],
        ).fetchone()
        if existing_id:
            self._conn.execute(
                """
                UPDATE manager_rookie_pick_profiles SET
                    computed_at = CURRENT_TIMESTAMP,
                    pick_premium_score = ?,
                    pick_trade_evidence = ?,
                    draft_selection_count = ?,
                    positional_tendency_json = ?,
                    dominant_archetype = ?,
                    archetype_pattern_json = ?,
                    show_draft_picks_tab = ?
                WHERE league_id = ? AND roster_id = ?
                """,
                [
                    profile.pick_premium_score,
                    profile.pick_trade_evidence,
                    profile.draft_selection_count,
                    json.dumps(profile.positional_tendency, separators=(",", ":")),
                    profile.dominant_archetype,
                    json.dumps(profile.archetype_pattern, separators=(",", ":")),
                    profile.show_draft_picks_tab,
                    profile.league_id,
                    profile.roster_id,
                ],
            )
        else:
            new_id = (
                self._conn.execute(
                    "SELECT COALESCE(MAX(id), 0) + 1 FROM manager_rookie_pick_profiles"
                ).fetchone()[0]
            )
            self._conn.execute(
                """
                INSERT INTO manager_rookie_pick_profiles
                    (id, league_id, roster_id, computed_at, pick_premium_score,
                     pick_trade_evidence, draft_selection_count, positional_tendency_json,
                     dominant_archetype, archetype_pattern_json, show_draft_picks_tab)
                VALUES (?, ?, ?, CURRENT_TIMESTAMP, ?, ?, ?, ?, ?, ?, ?)
                """,
                [
                    new_id,
                    profile.league_id,
                    profile.roster_id,
                    profile.pick_premium_score,
                    profile.pick_trade_evidence,
                    profile.draft_selection_count,
                    json.dumps(profile.positional_tendency, separators=(",", ":")),
                    profile.dominant_archetype,
                    json.dumps(profile.archetype_pattern, separators=(",", ":")),
                    profile.show_draft_picks_tab,
                ],
            )

    def upsert_draft_selection(self, sel: dict[str, Any]) -> None:
        """Upsert a single draft pick selection row.
        sel keys: league_id, draft_id, roster_id, player_id, pick_slot,
                  round_number, season, draft_type, position, archetype_label
        """
        existing_id = self._conn.execute(
            """SELECT id FROM draft_pick_selections
               WHERE league_id = ? AND draft_id = ? AND roster_id = ? AND player_id = ?""",
            [sel["league_id"], sel["draft_id"], sel["roster_id"], sel["player_id"]],
        ).fetchone()
        if existing_id:
            self._conn.execute(
                """UPDATE draft_pick_selections SET
                   pick_slot = ?, round_number = ?, season = ?, draft_type = ?,
                   position = ?, archetype_label = ?
                   WHERE league_id = ? AND draft_id = ? AND roster_id = ? AND player_id = ?""",
                [sel["pick_slot"], sel["round_number"], sel["season"], sel["draft_type"],
                 sel.get("position"), sel.get("archetype_label"),
                 sel["league_id"], sel["draft_id"], sel["roster_id"], sel["player_id"]],
            )
        else:
            new_id = self._conn.execute(
                "SELECT COALESCE(MAX(id), 0) + 1 FROM draft_pick_selections"
            ).fetchone()[0]
            self._conn.execute(
                """INSERT INTO draft_pick_selections
                   (id, league_id, draft_id, roster_id, player_id, pick_slot,
                    round_number, season, draft_type, position, archetype_label)
                   VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)""",
                [new_id, sel["league_id"], sel["draft_id"], sel["roster_id"],
                 sel["player_id"], sel["pick_slot"], sel["round_number"],
                 sel["season"], sel["draft_type"], sel.get("position"), sel.get("archetype_label")],
            )

    def get_draft_selections(self, league_id: str, roster_id: int) -> list[dict[str, Any]]:
        rows = self._conn.execute(
            """SELECT player_id, pick_slot, round_number, season, draft_type,
                      position, archetype_label
               FROM draft_pick_selections
               WHERE league_id = ? AND roster_id = ?
               ORDER BY season DESC, pick_slot ASC""",
            [league_id, roster_id],
        ).fetchall()
        cols = ["player_id", "pick_slot", "round_number", "season",
                "draft_type", "position", "archetype_label"]
        return [dict(zip(cols, row)) for row in rows]

    def get_all_league_selections(self, league_id: str) -> list[dict[str, Any]]:
        """All selections across all managers in a league (for league-relative positional tendency)."""
        rows = self._conn.execute(
            """SELECT roster_id, player_id, pick_slot, round_number, season,
                      draft_type, position, archetype_label
               FROM draft_pick_selections
               WHERE league_id = ?
               ORDER BY season DESC, pick_slot ASC""",
            [league_id],
        ).fetchall()
        cols = ["roster_id", "player_id", "pick_slot", "round_number", "season",
                "draft_type", "position", "archetype_label"]
        return [dict(zip(cols, row)) for row in rows]
```
